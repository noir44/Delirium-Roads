	/*
	Pallet of asphalt mix: 18 whole sacks on a vanilla pallet, roped down.
	AsphaltMix is base:drainable, so the input needs mode:destroy flags[ItemCount]
	- without them a count means USES, and the recipe would eat 18 charges out of
	one sack instead of taking eighteen sacks. IsFull keeps half-used sacks out:
	unloading hands back full ones.
	*/
	craftRecipe CreatePalletAsphaltMix
	{
		timedAction = Making,
		time = 150,
		category = RoadWorks,
		Tags = InHandCraft;CanBeDoneInDark,

		inputs
		{
			item 1 [Base.Mov_PalletEmpty],
			item 2 [Base.Rope],
			item 18 [DelRoads.AsphaltMix] mode:destroy flags[ItemCount;IsFull],
		}
		outputs
		{
			item 1 DelRoads.Mov_PalletAsphaltMix,
		}
	}

	craftRecipe OpenPalletAsphaltMix
	{
		timedAction = Making,
		time = 150,
		category = RoadWorks,
		Tags = InHandCraft;CanBeDoneInDark,

		inputs
		{
			item 1 [DelRoads.Mov_PalletAsphaltMix],
		}
		outputs
		{
			item 18 DelRoads.AsphaltMix,
			item 1 Base.Mov_PalletEmpty,
			item 2 Base.Rope,
		}
	}

	/*
	Pallets of bitumen (30 blocks) and gravel (18 sacks). Gravelbag is a
